#include<fstream>

void TaskManager::saveToFile(const std::string& filename) const {
   std::ofstream out(filename);
   for(const auto& task : task_) {
       out << task->getTitle() << "|" 
	       << static_cast<int>(task->getPriority()) << "|"
		   << static_cast<int>(task->getStatus()) << "\n";
   }
   std::cout << "Task saved to " <<  filename << '\n';
} 


void TaskManager::loadFromFile(const std::string& filename) {
  std::ifstream in(filename);
  std::string line;
  while(std::getline(in,line)) {
       auto sep1 = line.find('|');
	   auto sep2 = line.find('|',sep1 + 1); 
	   
	   std::string title = line.substr(0, sep1);
	   int priorityInt = std::stoi(line.substr(sep1+1, sep2-sep1-1));
	   int statusInt = std::stoi(line.substr(sep2 + 1)); 
	   
	   auto due = std::chrono::system_clock::now() + std::chrono::hours(24);
	   auto task = std::make_unique<Task>(title, due, static_cast<Priority>(priorityInt)); 
	   
	   if(static_cast<Status>(statusInt) == Status::Completed)
	     task->markCompleted();
		 
	   task_.push_back(std::move(task)); 
  }
 
  std::cout << "Tasks loaded from " << filename << "\n"; 
}
