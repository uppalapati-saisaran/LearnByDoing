Student_service.cpp (complete code below) 

#include "crow_all.h"
#include "json.hpp"
#include <unordered_map>
#include <atomic> 

using json = nlohmann::json;

//In-memory student DB 
std::unordered_map<int, json> students;
std::atomic<int> nextId{1};


int main() {
    crow::SimpleApp app;
	
	// POST/students 
	CROW_ROUTE(app,"/students").methods(crow::HTTPMethod::POST)([]const crow::request& req) {
	    auto body = json::parser(req.body);
		//Create student object 
		int id = nextId++;
		json student = {
		    {"id",id},
			{"name",body["name]},
			{"age",body["age"]}
		}; 
		students[id] = student;
		
		return crow::reponse{student.dump()};
	});
	
	// GET/students
	CROW_ROUTE(app,"/students").methods(corw::HTTPMethod::GET)([]() {
         json result = json::array();
		 for(const auto& [id,student] : students) {
		     result.push_back(student); 
		 }
		 return crow::response{result.dump(4)};  // Pretty JSON 
    )};	
	
	app.port(18080).multithreaded().run() 
}

//Helper for curl HTTP GET request 
// HTTP Client Helper: Receive response data into a string 
size_t writeCallback(void* contents, size_t size,size_t nmemb, std::string* out) {
    // contents is the data, size * nmemb is the actual number of bytes.
   
    size_t totalSize = size * nmemb; 
	// Calculates how many bytes were recived 
	out->append((char *)contents,totalSize);
	// Appends the received data to the output string(std::string* out). 
	return totalSize; 
	// Returns how many bytes were handled - required by curl 
}
// This is a callback function that libcurl calls whenever it receives part of the HTTP response.



// Helper Function to Make HTTP GET Request and Get JSON 

json httpGetJson(cont std::string& url) {
// This funciton takes a URL, makes a GET request to it, and returns the JSON response. 
    CURL* curl = curl_easy_init();
	// Initialize a new cURL session.
	std::string response;
	// We'll store the response from the HTTP server in this string. 
	if(curl) {
	// Check if initialization succeeded 
       curl_easy_setup(curl, CURLOPT_URL, url.c_str());
	   // Set the URL to request
	   curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, writeCallback);
	   // Set our earlier function to handle the incoming data. 
       curl_easy_setopt(curl, CURLOPT_WRITEDATA, &response);	
       // Pass the response string into the callback so data is saved into it. 	   
	}
	
	CURLcode res = curl_easy_perform(curl);
	// Perform the request store the result.
	curl_easy_cleanup(curl); 
	// Clean up the curl session 
	
	return json::parse(response,nullptr,false); 
	// Parse the raw response string as JSON 
	// if invalid, returns a null JSON object(because of false). 
}

Load Students Function 

function loadStudents() {
 // A function triggered when you click "Load Students".
 // It will fetch student data from the backend and display it.
 
 fetch('${API_BASE}/students') 
 // We send an HTTP GET request to /students on the backend.  
 
 .then(response => response.json()) 
 // When the server replies, we convert the response to JSON(JavaScript Object) 
 
 .then(data => {
 //Now that we have the data, we use it to update the table on the page. 
 
 let table = document.getElementById("StudentsTable');
 // We get a reference to the empty <table> element with ID studentsTable. 
  
 table.innerHTML = "<tr><th>ID</th><th>Name</th></tr>; 
 // This sets the header row for the table with column titles. 
 
 data.forEach(student => {
 // We loop over each student record from the server.
 
 table.innerHTML += '<tr><td>${student.id}</td><td>${student.name}</td></tr>'; 
 // For each student, we add a new row to the table showing their ID and Name 
 
 });
 });
 }
 //Done loading students. Entire function builds the student table dynamically.

Enroll Student in Course:

function enrollStudent() {
   const studentId = parseInt(document.getElementById("enrollStudentId").value);
   const courseId = parseInt(document.getElementById("enrollStudentId").value);
   // Gets the entered StudentID and CourseID from the input boxes.
   // Converts the values from string to number using parseInt(). 
   
   fetch('${API_BASE}/enroll'. {
       method: "POST",
	   headers: {"Content-Type":"application/json"},
	   body: JSON.stringify({studentId: studentId, courseId: courseId })
   })
   // Sends a POST to the /enroll endpoint with both IDs in the request body. 
   
   .then(response => {
    if(response.ok) {
	   alert("Enrollment successful");
    } else {
       alert("Enrollment failed.");
    }
   }); 

// If server replies with SUCCESS (200 OK), we show a success message. Otherwise, we show an error.
