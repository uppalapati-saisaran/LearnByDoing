Modify Employee Class 

Add to_dict() and from_dict() methods:

class Employee:
    def __init__(self, emp_id, name, department, role, email):
        self.emp_id = emp_id 
        self.name = name 
        self.department = department 
        self.role = role 
        self.email = email 
		
    def __str__(self):
        return f"{self.emp_id} - {self.name} | {self.role} ({self.department}) - {self.email}"
		
    def to_dict(self):
        return {
           "id": self.emp_id,
           "name": self.name,
           "department": self.department,
           "role": self.role,
           "email": self.email 
      }
	
    @staticmethod 
    def from_dict(data):
        return Employee(
        data["id"],
        data["name"],
        data["department"],
        data["role"],
        data["email"]
        )

def save_employees_to_file(employee_list, filename="data.json"):

def save_employees_from_file(filename="data.json"):
    try:
        with open(filename,"r") as f:
            data = json.load(f)
        return [Employee.from_dict(emp) for emp in data]
    except FileNotFoundError:
        return []

def group_by_department(employee_list):
    if not employee_list:
	    print("No employees to group.\n"
		return 
		
	department_map = {}
	
	for emp in employee_list:
	    if emp.department not in department_map:
		    department_map[emp.department] = []
		department_map[emp.department].append(emp) 
	
	for dept,emp_list in department_map.items():
	    print(f"\n {dept}")
	    for emp in emp_list:
		    print(f"- {emp.name} ({emp.role})")
